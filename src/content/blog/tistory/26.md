1\. v$active\_session\_history로 blocked\_session 파악

```sql
select sample_time
       ,session_id sid
       ,session_serial# serial
       ,session_type sestype
       ,user_id
       ,sql_id
       ,substr(sql_opname,1,6) opname
       ,substr(event,1,30) event
       ,wait_time
       ,session_state ses_state
       ,time_waited
       ,blocking_session blk_sess
       ,blocking_inst_id blk_inst#
       ,substr(module,1,30) module
       ,substr(program,1,20) program
       ,substr((SELECT OBJECT_NAME FROM DBA_OBJECTS C WHERE C.OBJECT_ID = a.CURRENT_OBJ#),1,25) object_name
from v$active_session_history a
where sample_time between to_date('2023/02/04 02:40','yyyy/mm/dd hh24:mi') and to_date('2023/02/04 03:00','yyyy/mm/dd hh24:mi')
order by sample_time, session_id;
```

2\. dba\_hist\_snapshot으로 execution / elapsedtime 출력

```sql
select to_char(end_interval_time,'yyyy-mm-dd HH24:MI') "Timestamp"
       ,sum(q.executions_delta) executions
       ,sum(q.buffer_gets_delta) buffer_gets
       ,sum(q.disk_reads_delta) disk_reads
       ,sum(q.rows_processed_delta) rows_pro
       ,round(sum(q.elapsed_time_delta)/1000000,1) "elapsed_time(S)"
       ,round(sum(q.cpu_time_delta)/1000000,1) "cpu_time(S)"
       ,round(decode(sum(q.buffer_gets_delta),0,1,sum(q.buffer_gets_delta))/decode(sum(q.executions_delta),0,1,sum(q.executions_delta))) buf_gets_exec
       ,round(decode(sum(q.disk_reads_delta),0,1,sum(q.disk_reads_delta))/decode(sum(q.executions_delta),0,1,sum(q.executions_delta))) disk_read_exec
       ,round(sum(q.elapsed_time_delta)/1000000/decode(sum(q.executions_delta),0,1,sum(q.executions_delta)),3) "Elap per exec(S)"
       ,round(sum(q.cpu_time_delta)/1000000/decode(sum(q.executions_delta),0,1,sum(q.executions_delta)),3) "CPU per exec(S)"
from dba_hist_snapshot s, dba_hist_sqlstat q
where s.snap_id = q.snap_id
and s.dbid = q.dbid
and s.instance_number = q.instance_number
and q.dbid = s.dbid
and q.instance_number = 1
--and q.sql_id = '7mzvwbga0ksvv'
and s.end_interval_time > sysdate - 15
group by to_char(end_interval_time,'yyyy-mm-dd HH24:MI')
order by 1;
```

3\. dba\_hist\_sqlstat & dba\_hist\_snapshot  수행속도 쿼리 추출

```sql
select to_char(begin_interval_time,'mm/dd/yyyy hh24:mi') dt
       ,dhss.instance_number ins
       ,plan_hash_value
       ,executions_delta
       ,rows_processed_delta
       ,sql_profile
       ,nvl(elapsed_time_delta/decode(executions_delta,0,null,executions_delta)/1000000,0) ela_exec
       ,round(nvl(elapsed_time_delta/decode(executions_delta,0,null,executions_delta)/1000000,0),2) ela_exec
       ,module
from dba_hist_sqlstat dhss, dba_hist_snapshot dhs
--where dhss.sql_id = '&sqlid'
where dhss.snap_id=dhs.snap_id
and dhss.instance_Number=dhs.instance_number
and begin_interval_time >= sysdate-15
order by begin_interval_time desc;
```
