> **내용**

> 각 테이블 별 인덱스 컬럼 확인하기가 어려워 한눈에 파티션 유무 파악 및 인덱스 LOCAL 유무등 파악할 수 있는 쿼리

> **SQL**

```sql
SELECT 
    T.OWNER,
    T.TABLE_NAME,
    I.INDEX_NAME,
    LISTAGG(I.COLUMN_NAME, ',') WITHIN GROUP (ORDER BY I.COLUMN_POSITION) AS INDEX_COLUMNS,
    T.TABLESPACE_NAME AS TABLE_DEF_TABLESPACE,
    T.PARTITIONED,
    TP.TABLESPACE_NAME AS PARTITION_TABLESPACE,
    --I.INDEX_OWNER,
    P.LOCALITY
FROM 
    DBA_TABLES T
    JOIN DBA_IND_COLUMNS I 
        ON T.OWNER = I.TABLE_OWNER AND T.TABLE_NAME = I.TABLE_NAME
    LEFT JOIN DBA_PART_INDEXES P 
        ON I.INDEX_OWNER = P.OWNER AND I.INDEX_NAME = P.INDEX_NAME
    LEFT JOIN (
        SELECT TABLE_OWNER, TABLE_NAME, MIN(PARTITION_POSITION) KEEP (DENSE_RANK FIRST ORDER BY PARTITION_POSITION) AS FIRST_PARTITION_POS,
               MIN(TABLESPACE_NAME) KEEP (DENSE_RANK FIRST ORDER BY PARTITION_POSITION) AS TABLESPACE_NAME
        FROM DBA_TAB_PARTITIONS
        GROUP BY TABLE_OWNER, TABLE_NAME
    ) TP 
        ON T.OWNER = TP.TABLE_OWNER AND T.TABLE_NAME = TP.TABLE_NAME
WHERE T.TABLE_NAME=':V_TABLE_NAME' --> :V_TABLE_NAME 테이블 이름 or
--WHERE T.OWNER=':V_OWNER' --> :V_OWNER OWNER
GROUP BY 
    T.OWNER,
    T.TABLE_NAME,
    T.TABLESPACE_NAME,
    T.PARTITIONED,
    TP.TABLESPACE_NAME,
    I.INDEX_OWNER,
    I.INDEX_NAME,
    P.LOCALITY;INDEX_NAME;
```
