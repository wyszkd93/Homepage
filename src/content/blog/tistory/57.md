```shell
SELECT A.MESSAGE
	 , A.START_TIME
	 , A.LAST_UPDATE_TIME
	 , A.TIME_REMAINING
	 , ROUND((SOFAR/TOTALWORK)*100,0)||'%' AS PROGESS
	 , B.SQL_TEXT
FROM V$SESSION_LONGOPS A, V$SQL B
WHERE A.SQL_ADDRESS=B.ADDRESS
--AND ROWNUM<=10
AND A.TIME_REMAINING>0;
```
