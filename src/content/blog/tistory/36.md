- 참고 컬럼 

   : snap이 찍혀있는지 확인 필요

## DBA\_HIST\_TBSPC\_SPACE\_USAGE

```sql
SELECT  A.*
      , CASE WHEN A.TABLESPACE_NAME IN ('SYSTEM', 'SYSAUX', 'TEMP') OR A.TABLESPACE_NAME LIKE 'UNDO%' 
                  THEN CASE WHEN USAGE > 80 THEN 'NOK'
                            ELSE 'OK'
                        END
             ELSE CASE WHEN A.RDATE < 10 AND "AVG-USED" > 0 AND "1-DAY" > 0 THEN 'NOK'
                       ELSE 'OK'
                   END
         END AS NOK
  FROM (  
          SELECT  TO_CHAR(sysdate, 'YYYYMMDD HH24:MI:SS') CTIME
                , DU.TABLESPACE_ID TBS_iD
                , TBS.TABLESPACE_NAME
                , TOT_SIZE_GB "TOT"
                , TOT_SIZE_GB - FREE_SIZE_GB "USED"
                , FREE_SIZE_GB "FREE"
                , TBS.USAGE "USAGE"
                , DU.DU1    "1-DAY"
                , DU.DU3    "3-DAY_AGV"
                , DU.DU7    "7-DAY_AVG"
                , DU.DUA    "M-DAY_AVG"
                , DU.DUM    "MAX-USED"
                , DU.DVAVG  "AVG-USED"
                , ROUND((FREE_SIZE_GB - DU.DUM) / (CASE WHEN DU.DVAVG  > 0  THEN DU.DVAVG
                                                         WHEN DU.DU7    > 0  THEN DU.DU7
                                                         WHEN DU.DU7    > 0  THEN DU.DU1
                                                         ELSE DECODE(ABS(DU.DVAVG), 0, 0.5, ABS(DU.DVAVG))
                                                     END)) AS RDATE
                , TBS.DBF_CNT
            FROM (
                    SELECT  TABLESPACE_ID
                          , TABLESPACE_NAME
                          , MAX(DECODE(LV, 1, AVG_USED_GB)) DU1
                          , MAX(DECODE(LV, 2, AVG_USED_GB)) DU3
                          , MAX(DECODE(LV, 3, AVG_USED_GB)) DU7
                          , MAX(DECODE(LV, 4, AVG_USED_GB)) DUA
                          , MAX(AVG_USED_GB) DUM
                          , ROUND(AVG(AVG_USED_GB), 2) DVAVG
                      FROM  (  
                               SELECT  TABLESPACE_ID
                                     , TABLESPACE_NAME
                                     , LV
                                     , ROUND(AVG(DAILY_USED_GB), 2) AVG_USED_GB
                                 FROM  (
                                          SELECT  A.TABLESPACE_ID
                                                , B.NAME TABLESPACE_NAME
                                                , A.RTIME
                                                , ROUND(A.TABLESPACE_USEDSIZE * C.DB_BLK_SZ / 1048576, 2)
                                                - ROUND(LAG(A.TABLESPACE_USEDSIZE, 1) OVER (PARTITION BY A.TABLESPACE_ID ORDER BY A.TABLESPACE_ID, RTIME) * C.DB_BLK_SZ / 1048576, 2) DAILY_USED_GB
                                            FROM  (  
                                                     SELECT  /*+ NO_MERGE */
                                                             TO_CHAR(TO_DATE(RTIME, 'MM/DD/YYYY HH24:MI:SS'), 'YYYYMMDD') RTIME
                                                           , A.TABLESPACE_ID
                                                           , A.TABLESPACE_USEDSIZE
                                                       FROM  DBA_HIST_TBSPC_SPACE_USAGE A
                                                      WHERE  1=1
                                                        AND  TO_CHAR(TO_DATE(RTIME, 'MM/DD/YYYY HH24:MI:SS'), 'HH24MI') = '0000'
                                                    ) A, 
                                                    V$TABLESPACE B, 
                                                    (SELECT VALUE/1024 DB_BLK_SZ FROM V$PARAMETER WHERE NAME = 'db_block_size') C
                                           WHERE  A.TABLESPACE_ID = B.TS#
                                        ) DU,
                                        (
                                           SELECT  TO_CHAR(TRUNC(SYSDATE) - DECODE(LEVEL, 1,1,2,3,3,7,4,32), 'YYYYMMDD') MIN_DATE, TO_CHAR(TRUNC(SYSDATE), 'YYYYMMDD') MAX_DATE, LEVEL LV
                                             FROM  DUAL
                                           CONNECT BY LEVEL <= 4
                                        ) DY
                                WHERE  DU.RTIME >= DY.MIN_DATE
                                  AND  DU.RTIME <  DY.MAX_DATE
                                GROUP BY TABLESPACE_ID, TABLESPACE_NAME, LV
                             )
                     GROUP BY TABLESPACE_ID, TABLESPACE_NAME                          
                  ) DU,
                  (  
                    SELECT  /*+  LEADING(D A F S) USE_HASH(D A F S) */
                            D.TABLESPACE_NAME,
                            DECODE(A.BYTES, NULL, 0, 0, 0, ROUND((NVL(A.BYTES, 0) - NVL(F.BYTES, 0)) / NVL(A.BYTES, 0) * 100, 2)) USAGE,
                            ROUND(NVL(A.BYTES, 0) / 1024 / 1024 / 1024, 2) TOT_SIZE_GB,
                            ROUND(NVL(F.BYTES, 0) / 1024 / 1024 / 1024, 2) FREE_SIZE_GB,
                            A.DBF_CNT
                      FROM  SYS.DBA_TABLESPACES D,
                            ( 
                               SELECT  TABLESPACE_NAME
                                     , SUM(BYTES) BYTES
                                     , SUM(DECODE(SIGN(BYTES - MAXBYTES), -1, MAXBYTES, BYTES)) MAXBYTES
                                     , COUNT(*) AS DBF_CNT
                                 FROM  SYS.DBA_DATA_FILES
                                GROUP BY TABLESPACE_NAME
                             ) A,
                             ( 
                                SELECT  TABLESPACE_ID
                                      , B.NAME
                                      , SUM_BLOCK * B.BLOCKSIZE BYTES
                                      , MAX_BLOCK * B.BLOCKSIZE MAX_BYTES
                                  FROM  (
                                           SELECT  TABLESPACE_ID
                                                 , SUM(SUM_BLOCK) SUM_BLOCK
                                                 , MAX(MAX_BLOCK) MAX_BLOCK
                                             FROM  (
                                                      SELECT  TABLESPACE_ID
                                                            , SUM(BLOCKS) SUM_BLOCK
                                                            , MAX(BLOCKS) MAX_BLOCK
                                                        FROM  SYS.DBA_LMT_FREE_SPACE       --데이터베이스에서 로컬로 관리되는 모든 테이블 스페이스의 여유 범위
                                                       GROUP BY TABLESPACE_ID
                                                       UNION ALL
                                                      SELECT  TABLESPACE_ID
                                                            , SUM(BLOCKS) SUM_BLOCK
                                                            , MAX(BLOCKS) MAX_BLOCK
                                                        FROM  SYS.DBA_DMT_FREE_SPACE        --데이터베이스의 모든 딕셔너리 관리 테이블 스페이스에 있는 여유 범위
                                                        GROUP BY TABLESPACE_ID 
                                                         UNION ALL
                                                       SELECT  TS# TABLESPACE_ID
                                                             , SUM(SPACE) SUM_BLOCK
                                                             , MAX(SPACE) MAX_BLOCK
                                                         FROM  SYS.RECYCLEBIN$              --DBA_RECYCLEBIN에는 block이 없음
                                                         GROUP BY TS#
                                                   )
                                            GROUP BY TABLESPACE_ID 
                                          ) A, SYS.TS$ B
                                 WHERE  A.TABLESPACE_ID = B.TS# 
                              ) F,
                              (
                                 SELECT  TS.NAME TABLESPACE_NAME
                                       , TS.TS#
                                       , SEG.MAX_EXT * TS.BLOCKSIZE MAX_EXT
                                   FROM  SYS.TS$ TS,
                                         (
                                            SELECT  TS#
                                                  , MAX(EXTSIZE) MAX_EXT
                                              FROM  SYS.SEG$
                                             GROUP BY TS#
                                          ) SEG
                                   WHERE  SEG.TS# = TS.TS# 
                               ) S
                     WHERE  D.TABLESPACE_NAME = A.TABLESPACE_NAME(+)
                       AND  D.TABLESPACE_NAME = F.NAME(+)
                       AND  D.TABLESPACE_NAME = S.TABLESPACE_NAME(+)
                       AND  NOT  (D.EXTENT_MANAGEMENT LIKE 'LOCAL' 
                                  AND D.CONTENTS LIKE 'TEMPORARY')
                    UNION ALL
                    SELECT  D.TABLESPACE_NAME
                          , DECODE(A.BYTES, NULL, 0, 0, 0, ROUND(NVL(T.BLOCKS, 0) * D.BLOCK_SIZE / NVL(A.BYTES, 0) * 100 / 2)) USAGE
                          , ROUND(NVL(A.BYTES, 0) / 1024 / 1024 / 1024, 2)
                          , ROUND((NVL(A.BYTES, 0) - NVL(T.BLOCKS, 0) * D.BLOCK_SIZE) / 1024 / 1024 / 1024, 2)
                          , A.DBF_CNT
                      FROM  SYS.DBA_TABLESPACES D,                        --block size와 tablespace 이름 추출을 위해 사용
                           (
                              SELECT  TABLESPACE_NAME
                                    , SUM(BYTES) BYTES                    --size of file(in Oracle blocks)
                                    , SUM(DECODE(SIGN(BYTES - MAXBYTES), -1, MAXBYTES, BYTES)) MAXBYTES  --sign함수 : 음수 판별, 인자가 < 0 이면 -1 리턴
                                    , COUNT(*) AS DBF_CNT
                                FROM  SYS.DBA_TEMP_FILES                 --데이터베이스의 모든 임시 파일을 설명
                               GROUP BY TABLESPACE_NAME
                            ) A,
                           (
                              SELECT  TABLESPACE TABLESPACE_NAME
                                    , SUM(BLOCKS) BLOCKS
                                FROM  SYS.V_$SORT_USAGE                   --모든 정렬 및 일부 내부 운영에 대한 내부 구성 요소
                               GROUP BY TABLESPACE
                            ) T
                     WHERE  D.TABLESPACE_NAME = A.TABLESPACE_NAME(+)
                       AND  D.TABLESPACE_NAME = T.TABLESPACE_NAME(+)
                       AND  D.EXTENT_MANAGEMENT LIKE 'LOCAL'
                       AND  D.CONTENTS LIKE 'TEMPORARY'
                   ) TBS
            WHERE  TBS.TABLESPACE_NAME = DU.TABLESPACE_NAME(+)
        ) A
ORDER BY TBS_ID;
```
